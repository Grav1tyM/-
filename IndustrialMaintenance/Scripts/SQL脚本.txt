/* ============================================================================
 * 工业维修智能化零件分配平台 —— MySQL 数据库初始化脚本
 * ----------------------------------------------------------------------------
 * 使用说明：
 *   1. 使用 MySQL 客户端（如 Navicat、MySQL Workbench、命令行）连接 MySQL
 *   2. 打开本脚本，直接执行
 *   3. 脚本会自动创建数据库 IndustrialMaintenance、全部数据表及演示种子数据
 *
 * 默认连接信息示例：
 *   Server=localhost; Database=IndustrialMaintenance; User=root; Password=你的密码
 *
 * 内置演示账号：
 *   admin    / admin123   （系统管理员，全部权限）
 *   engineer / 123456     （维修工程师，报修/工单/算料权限）
 *   store    / 123456     （仓库管理员，配件/领料权限）
 *
 * 说明：
 *   - 已按要求去掉密码加盐逻辑，Users 表不再包含 PasswordSalt 字段。
 *   - 演示账号密码以明文形式写入 PasswordHash 字段，便于直接测试；
 *     正式环境请改为程序端加密后存储。
 * ==========================================================================*/

/* --------------------------------------------------------------------------
 * 第 1 步：创建数据库（若不存在）
 * ------------------------------------------------------------------------*/
CREATE DATABASE IF NOT EXISTS IndustrialMaintenance
  DEFAULT CHARACTER SET utf8mb4
  DEFAULT COLLATE utf8mb4_general_ci;

USE IndustrialMaintenance;

/* --------------------------------------------------------------------------
 * 第 2 步：删除旧表（便于脚本重复执行，注意外键顺序：先删子表再删父表）
 * ------------------------------------------------------------------------*/
SET FOREIGN_KEY_CHECKS = 0;

DROP TABLE IF EXISTS MaterialRequirements;
DROP TABLE IF EXISTS RepairRecords;
DROP TABLE IF EXISTS WorkOrders;
DROP TABLE IF EXISTS BomRules;
DROP TABLE IF EXISTS Parts;
DROP TABLE IF EXISTS PartCategories;
DROP TABLE IF EXISTS Users;

SET FOREIGN_KEY_CHECKS = 1;

/* ==========================================================================
 * 第 3 步：创建数据表
 * ========================================================================*/

/* --------------------------------------------------------------------------
 * 3.1 用户表 Users
 *   - 存储系统登录账号
 *   - Role  角色：Admin=管理员 Engineer=维修工程师 Storekeeper=仓管员 User=普通用户
 *   - Status状态：1=启用 0=禁用（禁用后无法登录）
 *   - 已去掉 PasswordSalt 字段
 * ------------------------------------------------------------------------*/
CREATE TABLE Users
(
    Id            INT AUTO_INCREMENT PRIMARY KEY,           -- 用户ID（自增主键）
    UserName      VARCHAR(50)  NOT NULL UNIQUE,             -- 登录用户名（唯一）
    PasswordHash  VARCHAR(128) NOT NULL,                    -- 密码（去盐后直接存储，演示为明文）
    RealName      VARCHAR(50)  NULL,                        -- 真实姓名
    Role          VARCHAR(20)  NOT NULL DEFAULT 'User',     -- 角色（RBAC权限控制）
    Phone         VARCHAR(20)  NULL,                        -- 联系电话
    Status        INT          NOT NULL DEFAULT 1,          -- 状态：1启用 0禁用
    CreateTime    DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP -- 创建时间
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

/* --------------------------------------------------------------------------
 * 3.2 配件分类表 PartCategories
 * ------------------------------------------------------------------------*/
CREATE TABLE PartCategories
(
    Id           INT AUTO_INCREMENT PRIMARY KEY,          -- 分类ID
    CategoryName VARCHAR(50) NOT NULL UNIQUE,             -- 分类名称（如：电机类、轴承类）
    Remark       VARCHAR(200) NULL                        -- 备注
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

/* --------------------------------------------------------------------------
 * 3.3 配件信息表 Parts
 *   - PartCode 配件编码：与扫码枪扫描的条码一致，实现"扫一下自动带出配件"
 *   - Status 库存状态：InStock=在库充足 LowStock=库存不足 OutOfStock=缺货
 *     （状态由库存数量与最低库存 MinStock 比较得出，领料/入库时自动更新）
 * ------------------------------------------------------------------------*/
CREATE TABLE Parts
(
    Id            INT AUTO_INCREMENT PRIMARY KEY,          -- 配件ID
    PartCode      VARCHAR(50)  NOT NULL UNIQUE,            -- 配件编码（扫码枪扫描值）
    PartName      VARCHAR(100) NOT NULL,                   -- 配件名称
    Specification VARCHAR(100) NULL,                       -- 规格型号
    CategoryId    INT          NOT NULL,                   -- 所属分类ID（外键）
    Unit          VARCHAR(20)  NOT NULL DEFAULT '个',      -- 计量单位（个/套/件/升等）
    StockQuantity DECIMAL(12,2) NOT NULL DEFAULT 0,        -- 当前库存数量
    MinStock      DECIMAL(12,2) NOT NULL DEFAULT 0,        -- 最低库存预警值
    Price         DECIMAL(12,2) NULL DEFAULT 0,            -- 参考单价（元）
    Status        VARCHAR(20)  NOT NULL DEFAULT 'InStock', -- 库存状态
    Remark        VARCHAR(200) NULL,                       -- 备注
    CreateTime    DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP, -- 创建时间
    CONSTRAINT FK_Parts_Category FOREIGN KEY (CategoryId) REFERENCES PartCategories(Id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

/* --------------------------------------------------------------------------
 * 3.4 BOM 材料表（物料清单规则表）BomRules
 *   - 核心"智能算料"的依据：针对某个"维修部位/故障类型"，
 *     需要消耗哪些配件、每个单位用量多少、损耗率多少、最小领用量多少
 *   - RepairPart 维修部位/故障类型：如"电机""轴承""液压系统""电气线路"
 *   - QuantityPerUnit 单次维修基准用量（按 RepairQuantity=1 计算）
 *   - LossRate 损耗率（0.05 表示 5% 的装配损耗）
 *   - MinQuantity 最小领用量（即使算出的数量为0，也要领这么多，用于保护性更换）
 * ------------------------------------------------------------------------*/
CREATE TABLE BomRules
(
    Id              INT AUTO_INCREMENT PRIMARY KEY,          -- 规则ID
    RepairPart      VARCHAR(50)  NOT NULL,                   -- 维修部位/故障类型（与识别结果对应）
    PartId          INT          NOT NULL,                   -- 所需配件ID（外键）
    QuantityPerUnit DECIMAL(12,2) NOT NULL DEFAULT 1,        -- 每个维修单位的基准用量
    LossRate        DECIMAL(6,4)  NOT NULL DEFAULT 0,        -- 损耗率（0~1）
    MinQuantity     DECIMAL(12,2) NOT NULL DEFAULT 0,        -- 最小领用量
    Unit            VARCHAR(20)  NULL,                       -- 单位（冗余自配件表，便于直接展示）
    Remark          VARCHAR(200) NULL,                       -- 备注
    CONSTRAINT FK_BomRules_Part FOREIGN KEY (PartId) REFERENCES Parts(Id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

/* --------------------------------------------------------------------------
 * 3.5 维修工单表 WorkOrders
 *   - 报修 → 上传照片 → AI识别 → 智能算料 的流程载体
 *   - PhotoPath 维修部位照片的本地存储路径
 *   - RecognitionJson AI 识别结果（JSON 文本，保留原始识别证据）
 *   - Status 状态机：Pending待受理 → Processing处理中 → Completed已完成
 *             （Cancel 取消）
 *   - Priority 优先级：Normal=普通 Urgent=紧急
 * ------------------------------------------------------------------------*/
CREATE TABLE WorkOrders
(
    Id              INT AUTO_INCREMENT PRIMARY KEY,           -- 工单ID
    OrderNo         VARCHAR(30)   NOT NULL UNIQUE,            -- 工单号（如 WO20261009001）
    DeviceNo        VARCHAR(50)   NOT NULL,                   -- 设备编号
    DeviceName      VARCHAR(100)  NULL,                       -- 设备名称
    RepairPart      VARCHAR(50)   NULL,                       -- 维修部位（AI识别或人工填写）
    FaultDesc       VARCHAR(500)  NULL,                       -- 故障描述
    PhotoPath       VARCHAR(300)  NULL,                       -- 维修部位照片路径
    RecognitionJson LONGTEXT      NULL,                       -- AI识别结果JSON（原始证据）
    RepairQuantity  INT           NOT NULL DEFAULT 1,         -- 本次维修数量（台/套/个）
    Status          VARCHAR(20)   NOT NULL DEFAULT 'Pending', -- 工单状态
    Priority        VARCHAR(20)   NOT NULL DEFAULT 'Normal',  -- 优先级 Normal/Urgent
    Reporter        VARCHAR(50)   NULL,                       -- 报修人
    Assignee        VARCHAR(50)   NULL,                       -- 处理人（派工对象）
    AssignTime      DATETIME      NULL,                       -- 派工时间
    CreateTime      DATETIME      NOT NULL DEFAULT CURRENT_TIMESTAMP, -- 报修时间
    FinishTime      DATETIME      NULL,                       -- 完工时间
    Remark          VARCHAR(500)  NULL                        -- 备注
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

/* --------------------------------------------------------------------------
 * 3.6 领料清单表 MaterialRequirements
 *   - 智能算料的结果：一个工单对应多条领料记录
 *   - RequiredQuantity 系统按 BOM 规则推算出的应领数量
 *   - StockQuantity   当时库存快照（用于缺料判断展示）
 *   - IsShortage      是否缺料：1=库存不足 0=库存充足
 *   - Status 领料状态：Pending待领 Issued已领 Cancelled已取消
 * ------------------------------------------------------------------------*/
CREATE TABLE MaterialRequirements
(
    Id               INT AUTO_INCREMENT PRIMARY KEY,           -- 记录ID
    OrderNo          VARCHAR(30)   NOT NULL,                   -- 关联工单号
    PartId           INT           NOT NULL,                   -- 配件ID
    PartCode         VARCHAR(50)   NOT NULL,                   -- 配件编码
    PartName         VARCHAR(100)  NOT NULL,                   -- 配件名称
    Specification    VARCHAR(100)  NULL,                       -- 规格
    Unit             VARCHAR(20)   NULL,                       -- 单位
    RequiredQuantity DECIMAL(12,2) NOT NULL DEFAULT 0,         -- 推算应领数量
    StockQuantity    DECIMAL(12,2) NOT NULL DEFAULT 0,         -- 库存快照
    IsShortage       INT           NOT NULL DEFAULT 0,         -- 是否缺料 1缺 0不缺
    Status           VARCHAR(20)   NOT NULL DEFAULT 'Pending', -- 领料状态 Pending/Issued/Cancelled
    Remark           VARCHAR(200)  NULL,                       -- 备注（含算料明细说明）
    CreateTime       DATETIME      NOT NULL DEFAULT CURRENT_TIMESTAMP, -- 生成时间
    CONSTRAINT FK_Materials_Part FOREIGN KEY (PartId) REFERENCES Parts(Id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

/* --------------------------------------------------------------------------
 * 3.7 维修历史记录表 RepairRecords
 *   - 工单完工后归档维修历史，用于"维修频次"等统计分析
 * ------------------------------------------------------------------------*/
CREATE TABLE RepairRecords
(
    Id           INT AUTO_INCREMENT PRIMARY KEY,           -- 记录ID
    WorkOrderId  INT            NOT NULL,                  -- 关联工单ID
    DeviceNo     VARCHAR(50)    NOT NULL,                  -- 设备编号
    DeviceName   VARCHAR(100)   NULL,                      -- 设备名称
    RepairPart   VARCHAR(50)    NULL,                      -- 维修部位
    FaultDesc    VARCHAR(500)   NULL,                      -- 故障描述
    PartsSummary VARCHAR(500)   NULL,                      -- 使用的配件摘要
    Engineer     VARCHAR(50)    NULL,                      -- 维修工程师
    RepairTime   DATETIME       NOT NULL DEFAULT CURRENT_TIMESTAMP, -- 维修完成时间
    Remark       VARCHAR(200)   NULL                       -- 备注
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
